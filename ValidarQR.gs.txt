/**
 * ═══════════════════════════════════════════════════════
 *  REEMPLAZA la función validarQR en tu Code.gs
 *  por esta versión actualizada.
 *
 *  Cambios respecto a la versión anterior:
 *  - Devuelve dias, vencimiento y maquinaria
 *  - Formatea correctamente la fecha de vencimiento
 *  - Maneja fechas inválidas sin romper la respuesta
 * ═══════════════════════════════════════════════════════
 */
function validarQR(payload) {

  const sheet     = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("EQUIPO");
  const data      = sheet.getDataRange().getValues();
  const qrBuscado = String(payload.qr || "").trim();

  if (!qrBuscado) return { status: "ok", existe: false };

  // Determinar si es folio (col K) o código QR (col S)
  // Folio tiene formato XXXXX-YYYYY (5 dígitos - 5 dígitos)
  const esFolio = /^\d{5}-\d{5}$/.test(qrBuscado);

  for (let i = 1; i < data.length; i++) {

    let coincide = false;

    if (esFolio) {
      // Buscar por folio en columna K
      const folio = String(data[i][10] || "").trim();
      coincide = folio === qrBuscado;
    } else {
      // Buscar por código QR en columna S
      const codigoS = String(data[i][18] || "").trim();
      if (!codigoS) continue;
      coincide = codigoS === qrBuscado || qrBuscado.includes(codigoS);
    }

    if (!coincide) continue;

    const fechaRaw = data[i][8]; // Columna I

    let fechaStr       = "";
    let vencimientoStr = "";
    let diasRestantes  = null;
    let vigente        = false;

    try {
      const fecha = new Date(fechaRaw);
      if (!isNaN(fecha.getTime())) {
        const vencimiento = new Date(fecha);
        vencimiento.setFullYear(vencimiento.getFullYear() + 1);

        const hoy = new Date();
        hoy.setHours(0, 0, 0, 0);

        diasRestantes  = Math.floor((vencimiento - hoy) / 86400000);
        vigente        = diasRestantes >= 0;
        fechaStr       = Utilities.formatDate(fecha,      Session.getScriptTimeZone(), "dd/MM/yyyy");
        vencimientoStr = Utilities.formatDate(vencimiento, Session.getScriptTimeZone(), "dd/MM/yyyy");
      }
    } catch(e) {
      console.log("Error procesando fecha en fila " + (i+1) + ": " + e);
    }

    return {
      status:  "ok",
      existe:  true,
      vigente: vigente,
      dias:    diasRestantes,
      datos: {
        maquinaria:  String(data[i][3]  || ""),  // Col D
        marca:       String(data[i][4]  || ""),  // Col E
        modelo:      String(data[i][5]  || ""),  // Col F
        serie:       String(data[i][6]  || ""),  // Col G
        capacidad:   String(data[i][14] || ""),  // Col O  ← NUEVO
        fecha:       fechaStr,
        vencimiento: vencimientoStr
      }
    };
  }

  return { status: "ok", existe: false };
}
