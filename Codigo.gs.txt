/**
 * [APPSCRIPT] FUNCIÓN DOGET UNIFICADA
 * Determina si mostrar el Panel de Gestión o la Validación de QR
 */
function doGet(e) {

  // 🔹 1. VALIDACIÓN DE CERTIFICADOS (tu lógica original)
  if (e.parameter.id) {
    return validarCertificadoPublico(e.parameter.id);
  }

  // 🔹 2. API PARA FRONTEND (Hostinger)
  if (e.parameter.action) {

    let data;

    if (e.parameter.action === 'getMaquinaria') {
      data = obtenerOpcionesMaquinaria();
    } 
    else if (e.parameter.action === 'getChecklist') {
      data = obtenerChecklistPorEquipo(e.parameter.equipo);
    }
    else if (e.parameter.action === 'getDataCertificaciones') {
      data = obtenerDataCertificaciones();
    }
    else if (e.parameter.action === 'getDataCalidad') {        // ← ESTE FALTABA
  data = getDataCalidad();
}
  else if (e.parameter.action === "LISTAR_USUARIOS") {
    data = listarUsuarios();
  }


    else if (e.parameter.action === "IMPRIMIR_PDF") {
  data = imprimirPDF({
    fila: e.parameter.fila
  });
}

else if (e.parameter.action === "GENERAR_PDF") {
  data = generarPDFEnvio({
    fila: e.parameter.fila
  });
}
else if (e.parameter.action === "IMPRIMIR_PDF_CERT") {
  data = imprimirPDFCert({ fila: parseInt(e.parameter.fila) });
}

else if (e.parameter.action === "IMPRIMIR_PDF_DICT") {
  data = imprimirPDFDict({ fila: parseInt(e.parameter.fila) });
}

else if (e.parameter.action === "ENVIAR_CERTIFICADO") {
  data = enviarCertificado({
    fila: e.parameter.fila,
    fileId: e.parameter.fileId
  });
}

    else if (e.parameter.action === "VALIDAR_QR") {

  const result = validarQR({
    qr: e.parameter.qr || ""
  });

  return ContentService
    .createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}



    // 👉 respuesta JSON
    return ContentService
      .createTextOutput(JSON.stringify(data))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // 🔹 3. CARGA NORMAL DEL SISTEMA (si entras directo)
  return HtmlService
    .createTemplateFromFile('index')
    .evaluate()
    .setTitle("Sistema de Gestión AVBA")
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}


/**
 * [APPSCRIPT] FUNCIÓN DE VALIDACIÓN CON COLUMNAS CORREGIDAS
 */
function validarCertificadoPublico(idBusqueda) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.HOJA_EQUIPO);
  const data = sheet.getDataRange().getValues();
  
  // MAPEO DE COLUMNAS SEGÚN TU ESTRUCTURA REAL:
  const colControl = 10;   // Columna K (Folio)
  const colCliente = 1;    // Columna B (Cliente)
  const colMarca   = 3;    // Columna D (Marca)
  const colSerie   = 6;    // Columna G (Serie)
  const colFechaCert = 8;  // COLUMNA I (Fecha Certificación) - ¡CORREGIDO!

  const idLimpio = idBusqueda.trim().toUpperCase();
  const hoy = new Date();
  let registroEncontrado = null;

  for (let i = 1; i < data.length; i++) {
    let valorControl = data[i][colControl] ? data[i][colControl].toString().trim().toUpperCase() : "";
    
    if (valorControl === idLimpio) {
      let fechaCertRaw = data[i][colFechaCert];
      let fechaTexto = "Fecha no disponible";
      let vencimientoTexto = "No disponible";
      let estaVigente = false;

      // VALIDACIÓN DE FECHA EN COLUMNA I
      if (fechaCertRaw && (fechaCertRaw instanceof Date) && !isNaN(fechaCertRaw.getTime())) {
        let fechaCert = fechaCertRaw;
        let fechaVencimiento = new Date(fechaCert);
        fechaVencimiento.setFullYear(fechaVencimiento.getFullYear() + 1);
        
        fechaTexto = Utilities.formatDate(fechaCert, "GMT-6", "dd/MM/yyyy");
        vencimientoTexto = Utilities.formatDate(fechaVencimiento, "GMT-6", "dd/MM/yyyy");
        estaVigente = hoy <= fechaVencimiento;
      }

      registroEncontrado = {
        valido: true,
        control: valorControl,
        cliente: data[i][colCliente] || "N/A",
        marca: data[i][colMarca] || "N/A",
        serie: data[i][colSerie] || "N/A",
        fechaCertificacion: fechaTexto,
        fechaVencimiento: vencimientoTexto,
        vigente: estaVigente
      };
      break; 
    }
  }

  if (!registroEncontrado) {
    registroEncontrado = { valido: false };
  }

  const template = HtmlService.createTemplateFromFile('resultado_qr');
  template.datos = registroEncontrado;
  
  return template.evaluate()
      .setTitle("Validación AVBA - " + idLimpio)
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

/**
 * Versión para Web: Genera el PDF y retorna la URL para visualización inmediata.
 */
function imprimirPDFtemporal(filaWeb) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.HOJA_EQUIPO || 'EQUIPO');
  
  // Si viene de la web usa filaWeb, si se corre manual usa la celda activa
  const fila = filaWeb || sheet.getActiveCell().getRow();
  if (fila <= 1) return "Error: Fila no válida";

  // Usamos los IDs de tu objeto CONFIG o los definidos en la función
  const ID_PLANTILLA = "1Uwtiot9ickZh-nXmEu6xtGdyYfNIQRzWtdzCCak2wGI"; 
  const ID_CARPETA_PDF = CONFIG.ID_CARPETA_TEMPORALES || "1aW5Z3D3oC-ymHJTrK_gwjtLkVPV2gK1k";

  // Obtenemos los datos de la fila (Col A hasta T)
  const registro = sheet.getRange(fila, 1, 1, 20).getValues()[0];
  const formulaQR = sheet.getRange(fila, 19).getFormula(); // Columna S

  try {
    const datos = {
      folio:      registro[10], // K
      cliente:    registro[1],  // B
      domicilio:  registro[13], // N
      maquinaria: registro[3],  // D
      marca:      registro[4],  // E
      modelo:     registro[5],  // F
      serie:      registro[6],  // G
      idInterno:  registro[7],  // H
      fecha:      registro[8],  // I
      qrFormula:  String(formulaQR || "")
    };

    // Crear copia temporal del documento
    const copiaDoc = DriveApp.getFileById(ID_PLANTILLA).makeCopy(`TEMP_${datos.folio}`, DriveApp.getFolderById(ID_CARPETA_PDF));
    const doc = DocumentApp.openById(copiaDoc.getId());
    const body = doc.getBody();

    // Proceso de Reemplazo de Etiquetas
    body.replaceText("{{FOLIO}}", datos.folio || "S/F");
    body.replaceText("{{CLIENTE}}", datos.cliente || "");
    body.replaceText("{{Domicilio}}", datos.domicilio || "");
    body.replaceText("{{Maquinaria}}", datos.maquinaria || "");
    body.replaceText("{{Marca}}", datos.marca || "");
    body.replaceText("{{Modelo}}", datos.modelo || "");
    body.replaceText("{{Serie}}", datos.serie || "");
    body.replaceText("{{ID}}", datos.idInterno || "");
    
    const fTxt = datos.fecha instanceof Date ? Utilities.formatDate(datos.fecha, "GMT-6", "dd/MM/yyyy") : datos.fecha;
    body.replaceText("{{FECHA}}", fTxt || "");

    // Lógica para insertar la imagen del QR
    const busquedaQR = body.findText("{{QR}}");
    if (busquedaQR && datos.qrFormula.includes("http")) {
      const urlQR = datos.qrFormula.match(/"([^"]+)"/)[1];
      const img = UrlFetchApp.fetch(urlQR).getBlob();
      const elemento = busquedaQR.getElement();
      elemento.asText().setText("");
      elemento.getParent().asParagraph().appendInlineImage(img).setWidth(100).setHeight(100);
    }

    doc.saveAndClose();
    
    // Convertir a PDF
    const pdfBlob = copiaDoc.getAs(MimeType.PDF);
    const pdfFile = DriveApp.getFolderById(ID_CARPETA_PDF).createFile(pdfBlob);
    
    // Limpieza: Borrar el DOC temporal (mantener solo el PDF)
    DriveApp.getFileById(copiaDoc.getId()).setTrashed(true);

    const urlFinal = pdfFile.getUrl();
    
    // Opcional: Guardar el enlace en la Columna T (20) para registro
    sheet.getRange(fila, 20).setValue(urlFinal);
    
    return urlFinal; 

  } catch (e) {
    return "Error en el servidor: " + e.message;
  }
}

/**
 * Genera el número de control único (NumA-NumB).
 * Bloquea duplicados, registra clientes nuevos y dispara el QR en Columna S.
 * * @param {number} filaManual - Fila procesada por el disparador o manualmente.
 */
function generarControl(filaManual) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetEquipo = ss.getSheetByName(CONFIG.HOJA_EQUIPO || 'EQUIPO');
  const sheetClientes = ss.getSheetByName('Clientes_Registro');

  // 1. Identificar fila y verificar si ya existe un folio
  const filaDestino = filaManual || sheetEquipo.getActiveCell().getRow();
  if (filaDestino <= 1) return; // Ignorar encabezados

  const celdaControl = sheetEquipo.getRange(filaDestino, 11); // Columna K (Folio)
  
  // VERIFICACIÓN DE SEGURIDAD: No sobrescribir si ya hay un folio
  if (celdaControl.getValue() !== "") {
    console.log("Acción cancelada: La fila " + filaDestino + " ya tiene un folio asignado.");
    return; 
  }

  // 2. Obtener nombre del cliente (Columna B)
  const nombreCliente = sheetEquipo.getRange(filaDestino, 2).getValue();
  if (!nombreCliente) {
    ss.toast("Escribe el nombre del cliente en la Columna B primero.", "⚠️ Falta Dato", 3);
    return;
  }

  // 3. Obtener o registrar ID de Cliente (NumA)
  const clientesData = sheetClientes.getDataRange().getValues();
  let numA = "";
  let maxNumA = 0;

  for (let i = 1; i < clientesData.length; i++) {
    const idActual = parseInt(clientesData[i][1], 10);
    if (!isNaN(idActual) && idActual > maxNumA) maxNumA = idActual;
    
    // Normalización para evitar duplicados por minúsculas/mayúsculas
    if (clientesData[i][0].toString().trim().toUpperCase() === nombreCliente.toString().trim().toUpperCase()) {
      numA = clientesData[i][1].toString().padStart(5, '0');
    }
  }

  // Si el cliente es nuevo, se le asigna el siguiente ID global
  if (numA === "") {
    numA = (maxNumA + 1).toString().padStart(5, '0');
    sheetClientes.appendRow([nombreCliente, numA]);
  }

  // 4. Calcular el consecutivo de Equipo (NumB)
  const equipoData = sheetEquipo.getDataRange().getValues();
  let maxNumB = 0;

  for (let i = 1; i < equipoData.length; i++) {
    const valorK = equipoData[i][10]; // Columna K
    if (valorK && valorK.toString().includes('-')) {
      const partes = valorK.toString().split('-');
      const numBVal = parseInt(partes[1], 10);
      if (!isNaN(numBVal) && numBVal > maxNumB) maxNumB = numBVal;
    }
  }

  // 5. Construcción del Folio Final
  const nuevoNumB = (maxNumB + 1).toString().padStart(5, '0');
  const folioFinal = `${numA}-${nuevoNumB}`;

  // 6. Escritura de resultados y Formato
  celdaControl.setValue(folioFinal)
              .setHorizontalAlignment("center")
              .setFontWeight("bold")
              .setBackground("#f3f3f3"); // Gris tenue para indicar que es automático

  // 7. LANZAR GENERACIÓN DE QR EN COLUMNA S
  try {
    generarQR(filaDestino, folioFinal);
  } catch(e) {
    console.log("Error al generar QR: " + e);
    ss.toast("Folio creado, pero el QR falló.", "⚠️ Aviso", 3);
  }

  ss.toast("Folio: " + folioFinal + " vinculado correctamente.", "✅ AVBA Sistema", 3);
  return folioFinal;
}

  const COL_ESTADO_CALIDAD = 16;  // Columna P
  const COL_LINK_PDF = 20;        // Columna T
  const COL_REGISTRO_FECHA = 22;  // Columna V

  // IDs de Drive (Asegúrate de poner tus IDs reales aquí)
  const ID_PLANTILLA = "1Uwtiot9ickZh-nXmEu6xtGdyYfNIQRzWtdzCCak2wGI"; 
  const ID_CARPETA_FINAL = "1rS58dcGgsfgM3OeGg1z-oZ5LlZbGXOpT";

  }

    sheet.getRange(fila, COL_CANDADO_ENVIO).setValue("ENVIADO").setBackground("#d9ead3").setFontWeight("bold");

    return "EXITO";

  } catch (e) {
    Logger.log("Error: " + e.message);
    return "ERROR: " + e.message;
  }
}

/**
 * Busca o crea una carpeta para el cliente dentro de la carpeta raíz
 */
function obtenerCarpetaCliente(cliente, rootFolder) {
  const folders = rootFolder.getFoldersByName(cliente);
  if (folders.hasNext()) {
    return folders.next();
  }
  return rootFolder.createFolder(cliente);
}

  const fecha = new Date();
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];

  // CASO A: Edición de una sola celda
  if (numFilas === 1 && numColumnas === 1) {
    const encabezado = headers[colInicio - 1] || `Columna ${colInicio}`;
    sheetHistorial.appendRow([
      fecha, usuario, filaInicio, colInicio, encabezado, 
      e.oldValue || "(Vacío)", e.value || "(Eliminado/Vacío)"
    ]);
  } 
  // CASO B: Edición masiva (Pegado de datos o borrado de bloques)
  else {
    const nuevosValores = range.getValues();
    let registros = [];

    for (let i = 0; i < numFilas; i++) {
      for (let j = 0; j < numColumnas; j++) {
        let filaActual = filaInicio + i;
        let colActual = colInicio + j;
        if (filaActual === 1) continue; 

        let encabezado = headers[colActual - 1] || `Columna ${colActual}`;
        let valorNuevo = nuevosValores[i][j];

        registros.push([
          fecha, usuario, filaActual, colActual, encabezado, 
          "Cambio Masivo", valorNuevo === "" ? "(Borrado)" : valorNuevo
        ]);
      }
    }

    if (registros.length > 0) {
      sheetHistorial.getRange(sheetHistorial.getLastRow() + 1, 1, registros.length, 7).setValues(registros);
    }
  }
}

// 2. Función maestra para crear el PDF
// AÑADIMOS 'fotos' a los parámetros de la función
function generarPDF(info, checklist, idCarpeta, fotos) {
  let copiaId = null;
  try {
    const idPlantilla = obtenerIdPlantilla(info.maquinaria);
    if (!idPlantilla) return false;

    const nombreArchivo = "Reporte_" + info.serie;
    const carpetaDestino = DriveApp.getFolderById(idCarpeta);
    
    const archivoCopia = DriveApp.getFileById(idPlantilla).makeCopy(nombreArchivo, carpetaDestino);
    copiaId = archivoCopia.getId();
    const doc = DocumentApp.openById(copiaId);
    const body = doc.getBody();

    // --- 1. DATOS Y CHECKLIST ---
    body.replaceText("{{DIRECCION}}", info.direccion_inspeccion_legible || "");
    body.replaceText("{{TIPO_MAQUINARIA}}", info.maquinaria || "");
    body.replaceText("{{MARCA}}", info.marca || "");
    body.replaceText("{{MODELO}}", info.modelo || "");
    body.replaceText("{{SERIE}}", info.serie || "");
    body.replaceText("{{ID_EQUIPO}}", info.id_cliente || "");
    body.replaceText("{{FECHA}}", Utilities.formatDate(new Date(), "GMT-6", "dd/MM/yyyy"));
    body.replaceText("(?i){{comentarios}}", info.comentarios || "");

    if (checklist) {
      for (var clave in checklist) {
        var valor = checklist[clave];
        var texto = (valor === "C") ? "CONFORME" : (valor === "NC" ? "NO CONFORME" : valor);
        body.replaceText("(?i){{" + clave.replace(/[{}]/g, '') + "}}", texto);
      }
    }

// --- 3. INSERCIÓN DE FOTOS POR IDENTIFICADOR FIJO (F1 a F6) ---
// --- 3. INSERCIÓN DE FOTOS POR IDENTIFICADOR FIJO (F1 a F6) ---
    for (let i = 1; i <= 6; i++) {
      let etiqueta = "{{F" + i + "}}";
      let buscador = body.findText(etiqueta);
      
      if (buscador) {
        let elementoTexto = buscador.getElement();
        let celda = elementoTexto.getParent();
        
        while (celda && celda.getType() !== DocumentApp.ElementType.TABLE_CELL) {
          celda = celda.getParent();
        }

        if (celda) {
          let celdaTab = celda.asTableCell();
          celdaTab.clear(); 

          if (fotos && fotos[i - 1] && fotos[i - 1].base64) {
            try {
              let f = fotos[i - 1];
              let rawData = f.base64.replace(/^data:image\/(png|jpg|jpeg);base64,/, "").replace(/\s/g, "");
              let bytes = Utilities.base64Decode(rawData);
              let blob = Utilities.newBlob(bytes, f.mimeType || "image/jpeg");
              
              let img = celdaTab.appendImage(blob);
              let anchoRef = 170; 
              let ratio = anchoRef / img.getWidth();
              img.setWidth(anchoRef).setHeight(img.getHeight() * ratio);
              
              celdaTab.getChild(0).asParagraph().setAlignment(DocumentApp.HorizontalAlignment.CENTER);
            } catch (err) {
              console.error("Error en foto " + i + ": " + err.toString());
            }
          }
        }
      }
    }

    // --- 4. CIERRE Y CONVERSIÓN (FLUJO ÚNICO) ---
    // Paso 1: Guardar y forzar renderizado
    // Paso 1: Guardar y cerrar (Esto es lo que realmente asienta las fotos)
    doc.saveAndClose(); 
    
    // Paso 2: Pausa de seguridad para que el servidor procese el archivo
    Utilities.sleep(6000); 

    try {
      // Paso 3: Obtener el archivo y convertir a PDF
      var archivoDoc = DriveApp.getFileById(copiaId);
      
      // REFUERZO DEL NOMBRE: Nos aseguramos de que no venga vacío
      var serieFinal = (info && info.serie) ? info.serie : "S_N";
      var nombreFinalPdf = "Reporte_Inspeccion_" + serieFinal + ".pdf";

      var pdfBlob = archivoDoc.getAs(MimeType.PDF);
      pdfBlob.setName(nombreFinalPdf); // Asignamos el nombre reforzado
      
      carpetaDestino.createFile(pdfBlob);
      
      // Paso 4: Limpieza
      archivoDoc.setTrashed(true);
      return true;


    } catch (e) {
      console.error("Error en fase de PDF: " + e.toString());
      // Si falla el PDF, dejamos el Doc vivo para que no pierdas la info
      DriveApp.getFileById(copiaId).setName("REVISAR_IMAGENES_" + info.serie);
      return false;
    }
  } catch (e) {
    // Esto nos dirá exactamente qué línea y qué error es
    var errorDetallado = e.stack ? e.stack.split('\n')[0] : e.toString();
    console.error("Fallo crítico: " + errorDetallado);
    
    if (typeof copiaId !== 'undefined' && copiaId) {
      // El nombre del archivo ahora será el mensaje de error real
      var nombreError = ("ERR_" + errorDetallado).substring(0, 100);
      DriveApp.getFileById(copiaId).setName(nombreError);
    }
    return false;
  }
}

function getDataCalidad() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('EQUIPO');
  const data  = sheet.getDataRange().getValues();
  let resultados = [];

  for (let i = 1; i < data.length; i++) {
    // Ignorar filas vacías
    if (!data[i][1] && !data[i][6]) continue;

    try {
      resultados.push({
        fila:       i + 1,
        cliente:    String(data[i][1]  || ""),
        maquinaria: String(data[i][3]  || ""),
        marca:      String(data[i][4]  || ""),
        modelo:     String(data[i][5]  || ""),
        serie:      String(data[i][6]  || ""),
        id:         String(data[i][7]  || ""),
        fecha:      formatearFecha(data[i][8]),
        correo:     String(data[i][9]  || ""),
        control:    String(data[i][10] || ""),
        enviado:    String(data[i][11] || ""),
        evidencia:  String(data[i][12] || ""),
        direccion:  String(data[i][13] || ""),
        capacidad:  String(data[i][14] || ""),   // ← NUEVO: Columna O
        estado:     String(data[i][16] || "PENDIENTE"),
        motivo:     String(data[i][17] || ""),
        link:       String(data[i][19] || ""),
        dictamen:   String(data[i][20] || ""),
        envio:      String(data[i][21] || ""),
        
      });
    } catch(e) {
      console.log("Error en fila " + (i+1) + ": " + e.toString());
    }
  }

  return resultados;
}
function formatearFecha(fecha) {
  if (!fecha) return "";
  try {
    const f = new Date(fecha);
    if (isNaN(f.getTime())) return "";
    const dia = String(f.getDate()).padStart(2, '0');
    const mes = String(f.getMonth() + 1).padStart(2, '0');
    const anio = f.getFullYear();
    return `${dia}/${mes}/${anio}`;
  } catch(e) {
    return "";
  }
}

function actualizarCalidad(payload) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('EQUIPO');
  const fila  = payload.fila;

  if (payload.cliente)    sheet.getRange(fila, 2).setValue(payload.cliente);

  if (payload.marca)      sheet.getRange(fila, 5).setValue(payload.marca);
  if (payload.modelo)     sheet.getRange(fila, 6).setValue(payload.modelo);
  if (payload.serie)      sheet.getRange(fila, 7).setValue(payload.serie);
  if (payload.id)         sheet.getRange(fila, 8).setValue(payload.id);
  if (payload.fecha)      sheet.getRange(fila, 9).setValue(payload.fecha);
  if (payload.correo)     sheet.getRange(fila, 10).setValue(payload.correo);
  if (payload.direccion)  sheet.getRange(fila, 14).setValue(payload.direccion);

    sheet.getRange(fila, 17).setValue("NO CONFORME").setBackground("#ea9999");
    sheet.getRange(fila, 18).setValue(payload.motivo || "");
  }

  return { status: "success" };

function generarPDFBase(fila, templateId) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('EQUIPO');
  const data = sheet.getRange(fila, 1, 1, 23).getValues()[0];

  const copia = DriveApp.getFileById(templateId)
    .makeCopy(`TEMP_${data[10]}`);

  const doc = DocumentApp.openById(copia.getId());
  const body = doc.getBody();

  body.replaceText("{{FOLIO}}", data[10]);
  body.replaceText("{{CLIENTE}}", data[1]);
  body.replaceText("{{Domicilio}}", data[13]);
  body.replaceText("{{Maquinaria}}", data[3]);
  body.replaceText("{{Marca}}", data[4]);
  body.replaceText("{{Modelo}}", data[5]);
  body.replaceText("{{Serie}}", data[6]);
  body.replaceText("{{ID}}", data[7]);

  const fecha = Utilities.formatDate(new Date(data[8]), Session.getScriptTimeZone(), "dd/MM/yyyy");
  body.replaceText("{{FECHA}}", fecha);

  doc.saveAndClose();

  const pdf = copia.getAs(MimeType.PDF);
  copia.setTrashed(true);

  return pdf;
}

function generarPDFyEnviar() {

  const ss = SpreadsheetApp.getActive();
  const sheet = ss.getActiveSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data.shift();

  const rootFolder = DriveApp.getFolderById(ID_CARPETA_RAIZ);
  const tz = Session.getScriptTimeZone();

  const enviadoIndex = headers.indexOf('Enviado');

  const total = data.filter(row => row[enviadoIndex] === 'SI').length;
  let contador = 0;

  data.forEach((row, index) => {

    let record = {};
    headers.forEach((h, i) => {

      let value = row[i];

      if (value === '' || value === null) value = 'N/A';
      if (value instanceof Date) {
        value = Utilities.formatDate(value, tz, 'dd/MM/yyyy');
      }

      record[h] = value;
    });

    // Solo procesa los marcados como SI
    if (record['Enviado'] !== 'SI') return;
    if (record['Correo electronico'] === 'N/A') return;

    contador++;
    ss.toast(
      `Generando certificados ${contador} de ${total}`,
      'Progreso',
      3
    );

    // Generar Control automático
    if (record['Control'] === 'N/A') {
      const nuevoControl = generarControl(sheet, headers, index + 2);
      record['Control'] = nuevoControl;

      sheet.getRange(index + 2, headers.indexOf('Control') + 1)
        .setValue(nuevoControl);
    }

    // Carpeta por cliente
    const carpetaCliente = obtenerCarpetaCliente(
      record['Cliente o Razon social'],
      rootFolder
    );

    // Copiar plantilla
    const copy = DriveApp.getFileById(ID_PLANTILLA_DOC)
      .makeCopy(`Certificado_${record['Control']}`, carpetaCliente);

    const doc = DocumentApp.openById(copy.getId());
    const body = doc.getBody();

    for (let key in record) {
  let valor = record[key];

  if (typeof valor === 'string') {
    valor = valor.toUpperCase();
  }

  body.replaceText(`{{${key}}}`, valor);
}

    doc.saveAndClose();

    // Convertir a PDF
    const pdf = DriveApp.getFileById(copy.getId()).getAs(MimeType.PDF);

    // Enviar correo
    GmailApp.sendEmail(
  record['Correo electronico'],
  'Certificado de Inspección',
  `Estimado(a) cliente:

Reciba un cordial saludo.

Por medio del presente, hacemos llegar el Certificado de Inspección correspondiente a la evaluación realizada, el cual se adjunta en formato PDF para su conocimiento y resguardo.

El documento avala que el equipo inspeccionado cumple con los criterios técnicos y de seguridad aplicables, conforme a la revisión efectuada en la fecha indicada.

Agradecemos la confianza depositada en nuestros servicios. Para cualquier aclaración o información adicional, quedamos a sus órdenes.

Atentamente,

AVBA Inspections, Certifications and Maintenance S.A.S. de C.V.`,
  {
    attachments: [pdf],
    from: 'certificaciones@avba.com.mx',
    name: 'Depto AVBA Inspections'
  }
);

    // Marcar como ENVIADO
    sheet.getRange(index + 2, enviadoIndex + 1)
      .setValue('ENVIADO');

    // Registrar histórico
    ;
  });

  ss.toast(
    'Proceso finalizado correctamente',
    'Certificados',
    5
  );
}
