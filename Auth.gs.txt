const HOJA_USUARIOS = "USUARIOS";
const ROLES_VALIDOS = ["ADMIN","INSPECTOR","CALIDAD","CERTIFICACIONES","CLIENTE"];

function md5Hash(texto) {
  const raw = Utilities.computeDigest(
    Utilities.DigestAlgorithm.MD5, texto, Utilities.Charset.UTF_8
  );
  return raw.map(b => ('0' + (b & 0xff).toString(16)).slice(-2)).join('');
}

function inicializarHojaUsuarios() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(HOJA_USUARIOS);
  if (!sheet) {
    sheet = ss.insertSheet(HOJA_USUARIOS);
    sheet.appendRow(["usuario","password_hash","rol","nombre","id_cliente","activo","fecha_alta","ultimo_acceso"]);
    sheet.getRange("A1:H1").setFontWeight("bold").setBackground("#E6F1FB");
    sheet.setFrozenRows(1);
    // Crear admin por defecto: usuario=admin, password=avba2026
    const fecha = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm");
    sheet.appendRow(["admin", md5Hash("avba2026"), "ADMIN", "Administrador AVBA", "", true, fecha, ""]);
  }
  return sheet;
}

/* ── LOGIN ── */
function loginUsuario(payload) {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(HOJA_USUARIOS);
    if (!sheet) return { status:"error", message:"Hoja USUARIOS no encontrada." };

    const usuario = String(payload.usuario  || "").trim().toLowerCase();
    const pass    = String(payload.password || "").trim();
    if (!usuario || !pass) return { status:"error", message:"Usuario y contraseña requeridos." };

    const hash = md5Hash(pass);
    const data = sheet.getDataRange().getValues();

    for (let i = 1; i < data.length; i++) {
      const r = data[i];
      if (String(r[0]||"").trim().toLowerCase() !== usuario) continue;
      if (r[5] === false || String(r[5]).toUpperCase() === "FALSE")
        return { status:"error", message:"Esta cuenta está desactivada." };
      if (String(r[1]||"").trim().toLowerCase() !== hash)
        return { status:"error", message:"Contraseña incorrecta." };

      // Registrar acceso
      sheet.getRange(i+1, 8).setValue(
        Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm")
      );

      const token = md5Hash(usuario + Date.now());
      return {
        status:     "success",
        rol:        String(r[2]||"").toUpperCase(),
        nombre:     String(r[3]||""),
        usuario:    usuario,
        id_cliente: String(r[4]||""),
        token:      token
      };
    }
    return { status:"error", message:"Usuario no encontrado." };
  } catch(e) {
    return { status:"error", message:"Error interno: " + e.message };
  }
}

/* ── CRUD USUARIOS ── */
function crearUsuario(payload) {
  try {
    const sheet = inicializarHojaUsuarios();
    const usuario   = String(payload.usuario   ||"").trim().toLowerCase();
    const password  = String(payload.password  ||"").trim();
    const rol       = String(payload.rol       ||"").trim().toUpperCase();
    const nombre    = String(payload.nombre    ||"").trim();
    const idCliente = String(payload.id_cliente||"").trim();

    if (!usuario||!password||!rol||!nombre) return { status:"error", message:"Todos los campos son obligatorios." };
    if (!ROLES_VALIDOS.includes(rol)) return { status:"error", message:"Rol inválido." };
    if (rol === "CLIENTE" && !idCliente) return { status:"error", message:"id_cliente obligatorio para rol CLIENTE." };

    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]||"").trim().toLowerCase() === usuario)
        return { status:"error", message:"El usuario '" + usuario + "' ya existe." };
    }

    const fecha = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm");
    sheet.appendRow([usuario, md5Hash(password), rol, nombre, idCliente, true, fecha, ""]);
    return { status:"success", message:"Usuario creado correctamente." };
  } catch(e) {
    return { status:"error", message:e.message };
  }
}

function editarUsuario(payload) {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(HOJA_USUARIOS);
    if (!sheet) return { status:"error", message:"Hoja USUARIOS no encontrada." };

    const usuario = String(payload.usuario||"").trim().toLowerCase();
    const data    = sheet.getDataRange().getValues();

    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]||"").trim().toLowerCase() !== usuario) continue;
      const f = i + 1;
      if (payload.nombre     !== undefined) sheet.getRange(f,4).setValue(payload.nombre);
      if (payload.rol        !== undefined) sheet.getRange(f,3).setValue(String(payload.rol).toUpperCase());
      if (payload.id_cliente !== undefined) sheet.getRange(f,5).setValue(payload.id_cliente);
      if (payload.activo     !== undefined) sheet.getRange(f,6).setValue(payload.activo);
      if (payload.password) sheet.getRange(f,2).setValue(md5Hash(payload.password));
      return { status:"success" };
    }
    return { status:"error", message:"Usuario no encontrado." };
  } catch(e) {
    return { status:"error", message:e.message };
  }
}

function desactivarUsuario(payload) {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(HOJA_USUARIOS);
    if (!sheet) return { status:"error", message:"Hoja USUARIOS no encontrada." };
    const usuario = String(payload.usuario||"").trim().toLowerCase();
    const data    = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]||"").trim().toLowerCase() !== usuario) continue;
      sheet.getRange(i+1,6).setValue(false);
      return { status:"success" };
    }
    return { status:"error", message:"Usuario no encontrado." };
  } catch(e) {
    return { status:"error", message:e.message };
  }
}

function listarUsuarios() {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(HOJA_USUARIOS);
    if (!sheet) return [];
    const data = sheet.getDataRange().getValues();
    return data.slice(1).filter(r => r[0]).map((r, i) => ({
      fila:          i + 2,
      usuario:       String(r[0]||""),
      rol:           String(r[2]||""),
      nombre:        String(r[3]||""),
      id_cliente:    String(r[4]||""),
      activo:        r[5] === true || String(r[5]).toUpperCase() === "TRUE",
      fecha_alta:    String(r[6]||""),
      ultimo_acceso: String(r[7]||"")
    }));
  } catch(e) {
    return [];
  }
}

/* ── PORTAL CLIENTE ── */
function obtenerDatosCliente(payload) {
  try {
    const idCliente = String(payload.id_cliente||"").trim();
    if (!idCliente) return { status:"error", message:"id_cliente requerido." };

    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("EQUIPO");
    const data  = sheet.getDataRange().getValues();
    const equipos = [];

    for (let i = 1; i < data.length; i++) {
      const fila   = data[i];
      const folio  = String(fila[10]||"").trim();
      const estado = String(fila[16]||"").trim();
      if (!folio) continue;
      if (folio.split("-")[0] !== idCliente) continue;
      if (estado !== "ENVIADO") continue;

      let fechaStr = "", vencStr = "", dias = null;
      try {
        const f = new Date(fila[8]);
        if (!isNaN(f.getTime())) {
          const v = new Date(f); v.setFullYear(v.getFullYear()+1);
          fechaStr = Utilities.formatDate(f, Session.getScriptTimeZone(), "dd/MM/yyyy");
          vencStr  = Utilities.formatDate(v, Session.getScriptTimeZone(), "dd/MM/yyyy");
          const hoy = new Date(); hoy.setHours(0,0,0,0);
          dias = Math.floor((v - hoy) / 86400000);
        }
      } catch(e) {}

      const codigoQR = String(fila[18]||"").trim();
      equipos.push({
        folio:      folio,
        maquinaria: String(fila[3]||""),
        marca:      String(fila[4]||""),
        modelo:     String(fila[5]||""),
        serie:      String(fila[6]||""),
        id_equipo:  String(fila[7]||""),
        fecha:      fechaStr,
        vencimiento:vencStr,
        dias:       dias,
        link_cert:  String(fila[19]||""),
        link_dict:  String(fila[20]||""),
        qr_url:     codigoQR ? "https://quickchart.io/qr?text=" + encodeURIComponent(codigoQR) + "&size=300" : "",
        vigente:    dias !== null && dias >= 0
      });
    }

    // Info del cliente (nombre del primer registro)
    let nombreCliente = "";
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][10]||"").split("-")[0] === idCliente) {
        nombreCliente = String(data[i][1]||"");
        break;
      }
    }

    return { status:"success", nombre_cliente: nombreCliente, equipos };
  } catch(e) {
    return { status:"error", message:e.message };
  }
}
