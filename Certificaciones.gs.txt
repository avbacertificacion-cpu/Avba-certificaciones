/* ═══════════════════════════════════════════════════════
   FUNCIÓN AUXILIAR — FORMATEAR FECHA
═══════════════════════════════════════════════════════ */
function formatearFechaGs(fecha) {
  if (!fecha) return "";
  try {
    const f = new Date(fecha);
    if (isNaN(f.getTime())) return String(fecha);
    return Utilities.formatDate(f, Session.getScriptTimeZone(), "dd/MM/yyyy");
  } catch(e) {
    return String(fecha || "");
  }
}

/* ═══════════════════════════════════════════════════════
   FUNCIÓN AUXILIAR — OBTENER URL DEL QR
   Busca el código de 10 dígitos de col S (EQUIPO) en
   col A de la hoja QR, y extrae la URL de la fórmula
   =IMAGE("https://quickchart.io/qr?text=CODIGO&size=300")
   que está en col B para hacer fetch del blob.
═══════════════════════════════════════════════════════ */
function obtenerImagenQR(codigoQR) {
  const codigo = String(codigoQR || "").trim();
  if (!codigo) return null;
  try {
    // La URL sigue el mismo patron que la formula =IMAGE(...) en la hoja QR
    const url      = "https://quickchart.io/qr?text=" + encodeURIComponent(codigo) + "&size=300";
    const response = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
    if (response.getResponseCode() === 200) {
      const blob = response.getBlob();
      blob.setName("qr_" + codigo + ".png");
      return blob;
    }
    console.log("HTTP " + response.getResponseCode() + " al obtener QR: " + codigo);
    return null;
  } catch(e) {
    console.log("obtenerImagenQR error: " + e);
    return null;
  }
}


function obtenerUrlQR(codigoQR) {
  const codigo = String(codigoQR || "").trim();
  if (!codigo) return "";
  // Construir directamente con el mismo patrón de la fórmula =IMAGE(...)
  return "https://quickchart.io/qr?text=" + encodeURIComponent(codigo) + "&size=300";
}

/* ═══════════════════════════════════════════════════════
   FUNCIÓN CORE — GENERAR DOCUMENTO
   Crea una copia de la plantilla, reemplaza etiquetas
   e inserta la imagen QR. Devuelve el ID del documento.
═══════════════════════════════════════════════════════ */
function generarDocumento(idPlantilla, nombreArchivo, etiquetas, blobQR, idCarpeta) {
  const carpeta = DriveApp.getFolderById(idCarpeta);
  const copia   = DriveApp.getFileById(idPlantilla).makeCopy(nombreArchivo, carpeta);
  const doc     = DocumentApp.openById(copia.getId());
  const body    = doc.getBody();

  // Reemplazar todas las etiquetas de texto
  for (const etiqueta in etiquetas) {
    const valor = etiquetas[etiqueta];
    body.replaceText(etiqueta, valor !== null && valor !== undefined ? String(valor) : "");
  }

  // Insertar imagen QR — busca {{QR}} en párrafos Y en celdas de tablas
  if (blobQR) {
    let insertado = false;

    // Estrategia 1: buscar en párrafos normales
    const busquedaDirecta = body.findText("\{\{QR\}\}");
    if (busquedaDirecta) {
      try {
        const el     = busquedaDirecta.getElement();
        const parr   = el.getParent().asParagraph();
        el.asText().setText("");
        const img = parr.appendInlineImage(blobQR);
        img.setWidth(120).setHeight(120);
        insertado = true;
        console.log("QR insertado en párrafo normal");
      } catch(e) {
        console.log("Fallo párrafo normal: " + e);
      }
    }

    // Estrategia 2: buscar en tablas (celda por celda)
    if (!insertado) {
      const numTablas = body.getNumChildren();
      for (let t = 0; t < numTablas; t++) {
        const elemento = body.getChild(t);
        if (elemento.getType() !== DocumentApp.ElementType.TABLE) continue;
        const tabla = elemento.asTable();
        for (let r = 0; r < tabla.getNumRows(); r++) {
          const fila = tabla.getRow(r);
          for (let c = 0; c < fila.getNumCells(); c++) {
            const celda = fila.getCell(c);
            const texto = celda.getText();
            if (!texto.includes("{{QR}}")) continue;
            try {
              // Limpiar contenido de la celda
              celda.clear();
              // Agregar párrafo con la imagen
              const parr = celda.addParagraph("");
              parr.appendInlineImage(blobQR).setWidth(120).setHeight(120);
              parr.setAlignment(DocumentApp.HorizontalAlignment.CENTER);
              insertado = true;
              console.log("QR insertado en tabla [" + r + "][" + c + "]");
              break;
            } catch(e) {
              console.log("Fallo celda tabla: " + e);
            }
          }
          if (insertado) break;
        }
        if (insertado) break;
      }
    }

    if (!insertado) {
      // Último recurso: limpiar la etiqueta si no se pudo insertar
      body.replaceText("\{\{QR\}\}", "");
      console.log("QR: etiqueta limpiada (no se encontró ubicación válida)");
    }

  } else {
    body.replaceText("\{\{QR\}\}", "");
    console.log("QR: sin blob disponible, etiqueta limpiada");
  }

  doc.saveAndClose();
  return copia.getId();
}

/* ═══════════════════════════════════════════════════════
   FUNCIÓN — GENERAR PDF DESDE DOCUMENTO
═══════════════════════════════════════════════════════ */
function docAPDF(docId, nombrePDF, idCarpetaDestino) {
  Utilities.sleep(3000);
  const archivo = DriveApp.getFileById(docId);
  const pdfBlob = archivo.getAs(MimeType.PDF);
  pdfBlob.setName(nombrePDF + ".pdf");
  const carpeta = DriveApp.getFolderById(idCarpetaDestino);
  const pdfFile = carpeta.createFile(pdfBlob);

  // FIX: permisos públicos para que el link abra sin pedir cuenta de Google
  pdfFile.setSharing(
    DriveApp.Access.ANYONE_WITH_LINK,
    DriveApp.Permission.VIEW
  );

  archivo.setTrashed(true);
  return pdfFile;
}

/* ═══════════════════════════════════════════════════════
   ACCIÓN — IMPRIMIR PDF CERTIFICADO (vista previa)
═══════════════════════════════════════════════════════ */
function imprimirPDFCert(payload) {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("EQUIPO");
    const fila  = payload.fila;
    const datos = sheet.getRange(fila, 1, 1, 23).getValues()[0];

    const etiquetas = obtenerEtiquetas(datos);
    const valorQR   = sheet.getRange(fila, COL.QR).getFormula() || datos[18];
    const blobQR    = obtenerImagenQR(String(datos[18] || "").trim());

    const docId  = generarDocumento(
      PLANTILLAS.CERT_IMPRESION,
      "IMPRESION_CERT_" + (datos[10] || fila),
      etiquetas,
      blobQR,
      CARPETAS_CERT.TEMPORALES
    );

    const pdf = docAPDF(docId, "Certificado_Impresion_" + (datos[10] || fila), CARPETAS_CERT.TEMPORALES);

    return { status: "success", url: pdf.getUrl() };
  } catch(e) {
    console.error("imprimirPDFCert: " + e);
    return { status: "error", message: e.message };
  }
}

/* ═══════════════════════════════════════════════════════
   ACCIÓN — GENERAR CERTIFICADO Y ENVIAR POR CORREO
═══════════════════════════════════════════════════════ */
function generarCertEnviar(payload) {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("EQUIPO");
    const fila  = payload.fila;
    const datos = sheet.getRange(fila, 1, 1, 23).getValues()[0];

    const correo = String(datos[COL.CORREO - 1] || "").trim();
    if (!correo) return { status: "error", message: "El registro no tiene correo registrado" };

    const folio     = String(datos[COL.CONTROL - 1] || fila);
    const etiquetas = obtenerEtiquetas(datos);
    const blobQR    = obtenerImagenQR(String(datos[18] || "").trim());

    // Generar documento y PDF
    const docId = generarDocumento(
      PLANTILLAS.CERT_ENVIO,
      "Certificado_" + folio,
      etiquetas,
      blobQR,
      CARPETAS_CERT.CERTIFICADOS
    );

    Utilities.sleep(3000);
    const archivo  = DriveApp.getFileById(docId);
    const pdfBlob  = archivo.getAs(MimeType.PDF);
    pdfBlob.setName("Certificado_AVBA_" + folio + ".pdf");
    const pdfFile  = DriveApp.getFolderById(CARPETAS_CERT.CERTIFICADOS).createFile(pdfBlob);
    pdfFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    archivo.setTrashed(true);

    // Enviar correo
    MailApp.sendEmail({
      to:          correo,
      subject:     "Certificado de Inspección AVBA — Folio " + folio,
      htmlBody:    plantillaCorreo(datos[1], folio, "certificado"),
      attachments: [pdfBlob],
      from:        "certificaciones@avba.com.mx",
      name:        "AVBA Inspections"
    });

    // Guardar URL en columna T
    const fechaEnvio = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm");
    sheet.getRange(fila, COL.LINK_CERT).setValue(pdfFile.getUrl());
    sheet.getRange(fila, COL.ESTADO).setValue("ENVIADO").setBackground("#cce5ff");
    sheet.getRange(fila, COL.ENVIADO).setValue(fechaEnvio);

    return { status: "success", url: pdfFile.getUrl() };
  } catch(e) {
    console.error("generarCertEnviar: " + e);
    return { status: "error", message: e.message };
  }
}

/* ═══════════════════════════════════════════════════════
   OBTENER DATA PARA EL PANEL
   Solo registros con estado APROBADO CALIDAD o ENVIADO
═══════════════════════════════════════════════════════ */
function obtenerDataCertificaciones() {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("EQUIPO");
  const data  = sheet.getDataRange().getValues();
  const resultado = [];

  for (let i = 1; i < data.length; i++) {
    const fila   = data[i];
    const estado = String(fila[COL.ESTADO - 1] || "").trim();

    // Solo mostrar registros aprobados o enviados
    if (estado !== "APROBADO CALIDAD" && estado !== "ENVIADO") continue;
    if (!fila[1] && !fila[6]) continue; // Ignorar filas vacías

    try {
      // Obtener URL del QR usando el código de col S buscado en hoja QR col A
      const codigoQR = String(fila[18] || "").trim(); // Col S índice 18 (columna S)
      // DIAGNÓSTICO: descomenta esta línea si el QR no aparece en el panel
      // console.log("Fila " + (i+1) + " — codigoQR: [" + codigoQR + "]");
      const qrUrl    = codigoQR ? obtenerUrlQR(codigoQR) : "";
      // console.log("Fila " + (i+1) + " — qrUrl: " + qrUrl);

      resultado.push({
        fila:       i + 1,
        cliente:    String(fila[1]  || ""),
        maquinaria: String(fila[3]  || ""),
        marca:      String(fila[4]  || ""),
        modelo:     String(fila[5]  || ""),
        serie:      String(fila[6]  || ""),
        id:         String(fila[7]  || ""),
        fecha:      formatearFechaGs(fila[8]),
        correo:     String(fila[9]  || ""),
        control:    String(fila[10] || ""),
        enviado:    String(fila[11] || ""),
        evidencia:  String(fila[12] || ""),
        direccion:  String(fila[13] || ""),
        estado:     estado,
        motivo:     String(fila[17] || ""),
        qr_codigo:  codigoQR,              // Código de 10 dígitos col S
        qr_url:     qrUrl,                 // URL construida desde hoja QR
        link:       String(fila[19] || ""),   // Col T — certificado
        dictamen:   String(fila[20] || ""),   // Col U — dictamen
        envio:      String(fila[21] || ""),   // Col V — dirección envío
        coordenadas:String(fila[22] || ""),   // Col W — coordenadas
      });
    } catch(e) {
      console.log("Error en fila " + (i+1) + ": " + e.toString());
    }
  }

  return resultado;
}

/* ═══════════════════════════════════════════════════════
   ACCIÓN — GUARDAR DIRECCIÓN DE ENVÍO DESDE CERTIFICACIONES
═══════════════════════════════════════════════════════ */
function guardarEnvioCert(payload) {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("EQUIPO");
    const fila  = payload.fila;

    if (payload.envio !== undefined) {
      sheet.getRange(fila, COL.ENVIO_DIR).setValue(payload.envio);
    }
    if (payload.coordenadas !== undefined) {
      sheet.getRange(fila, COL.COORDENADAS).setValue(payload.coordenadas);
    }

    return { status: "success" };
  } catch(e) {
    console.error("guardarEnvioCert: " + e);
    return { status: "error", message: e.message };
  }
}

/* ═══════════════════════════════════════════════════════
   LIMPIEZA AUTOMÁTICA DE TEMPORALES
   Programar como disparador diario
═══════════════════════════════════════════════════════ */
function limpiarTemporalesCertificaciones() {
  const carpeta  = DriveApp.getFolderById(CARPETAS_CERT.TEMPORALES);
  const archivos = carpeta.getFiles();
  const limite   = 2 * 60 * 60 * 1000; // 2 horas
  const ahora    = new Date().getTime();

  while (archivos.hasNext()) {
    const archivo = archivos.next();
    if (ahora - archivo.getDateCreated().getTime() > limite) {
      archivo.setTrashed(true);
      console.log("Eliminado temporal: " + archivo.getName());
    }
  }
}
